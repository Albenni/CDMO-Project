\section{CP Model}
\label{sec:cp-model}

We model Sports Tournament Scheduling with two MiniZinc variants: a
\emph{decision} model (\texttt{sts\_decision.mzn}) and an \emph{optimization}
model (\texttt{sts\_opt.mzn}). Both use
\[
  \text{Teams} = \{1,\dots,n\},\quad
  \text{Weeks} = \{1,\dots,n-1\},\quad
  \text{Periods} = \{1,\dots,n/2\}.
\]
In this project we consider instances with an even number of teams; moreover,
the period–usage specification makes the case $n=4$ infeasible, so it is
excluded.

\noindent\textit{Notation.} Iverson brackets $[\,\cdot\,]$ denote Boolean$\to$integer casts.

\subsection{Decision variables}
\label{subsec:cp-vars}

\paragraph{Decision model}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \textbf{\(\bm{Z}\) (match-at-slot indicator):}
        \(Z_{w,p,i,j}\in\{0,1\}\) indicates whether the unordered pair
        \(\{i,j\}\) (\(i<j\)) is scheduled in slot \((w,p)\); this is the
        authoritative timetable view.
  \item \textbf{\(\bm{U}\) (team-at-slot indicator):}
        \(U_{w,p,t}\in\{0,1\}\) marks whether team \(t\) plays in \((w,p)\);
        it is linked to \(Z\) so weekly and per–period counts are stated
        directly.
  \item \textbf{\(\bm{H},\bm{A}\) (output matrices):}
        Period\(\times\)week arrays required by the checker, derived from the
        internal variables.
  \item \textbf{\(\bm{C}\) (exact per–period profile, see
        \emph{Implied constraints}):}
        \(C_{t,p}\in\{1,2\}\) counts the uses of period \(p\) by team \(t\).
\end{itemize}

\paragraph{Optimization model}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \(\bm{Z},\ \bm{U},\ (\bm{H},\bm{A}),\ \bm{C}\) (as above).
  \item \textbf{\(\bm{W}\) (pair–to–week mapping):}
        \(W_{i,j}\in\text{Weeks}\) for \(i<j\) records in which week the match
        \(\{i,j\}\) is played.
  \item \textbf{\(\bm{B}\) (orientation flag):}
        \(B_{i,j}\in\{0,1\}\) for \(i<j\) selects the home/away direction for
        the pair \(\{i,j\}\).
  \item \textbf{\(\bm{Hc},\ \bm{\mathrm{dev}},\ \mathrm{obj}\) (fairness
        counters and objective):}
        \(Hc_t\) is the number of home matches of team \(t\);
        \(\mathrm{dev}_t\) is the absolute deviation of team \(t\) from a
        perfectly balanced split; the objective
        \(\mathrm{obj}=\sum_{t}\mathrm{dev}_t\) aggregates the team-wise
        imbalances and is minimized.
\end{itemize}

\subsection{Objective function}
\label{subsec:cp-objective}

We minimize the total home/away imbalance across teams:
\[
  \min\ \mathrm{obj} = \sum_{t\in\text{Teams}} \mathrm{dev}_t,
\]
where \(\mathrm{dev}_t\) is the absolute difference between team \(t\)'s number
of home games and the ideal split \((n-1)/2\). Since each team plays \(n-1\)
matches (odd), the smallest per–team imbalance is \(1\); accordingly, we post
the pruning bounds
\[
  n \ \le\ \mathrm{obj}\ \le\ 2n,
\]
which tighten search without restricting feasibility.

\subsection{Constraints}
\label{subsec:cp-constraints}

\paragraph{Core constraints (both models, \(Z/U\))}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \textbf{One match per slot:}
        \(\forall w,p:\ \sum_{i<j} Z_{w,p,i,j}=1\).
  \item \textbf{Each unordered pair once:}
        \(\forall i<j:\ \sum_{w,p} Z_{w,p,i,j}=1\).
  \item \textbf{Weekly participation:}
        \(\forall w,t:\ \sum_{p} U_{w,p,t}=1\).
  \item \textbf{Period cap:}
        \(\forall t,p:\ \sum_{w} U_{w,p,t}\le 2\).
\end{itemize}

\paragraph{Index conventions (unordered pairs)}
We encode each unordered pair only once and disable meaningless entries:
\[
  \forall\,w,p,\ \forall\,i\ge j:\quad Z_{w,p,i,j}=0,
\]
\[
  \text{(optimization)}\qquad
  \forall\,i\ge j:\quad B_{i,j}=0,\quad W_{i,j}=1.
\]
These fixed cells are never ranged over (all sums use \(i<j\)); they simply
prevent spurious variables from appearing in the arrays.

\paragraph{Channeling}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \textbf{\(U\leftrightarrow Z\):}
        \(\forall w,p,t:\ 
          U_{w,p,t}=\sum_{j>t} Z_{w,p,t,j}+\sum_{i<t} Z_{w,p,i,t}\).
  \item \textbf{\(W\leftrightarrow Z\) (optimization):}
        \(\forall i<j,\forall w:\ 
          [\,W_{i,j}=w\,]=\sum_{p} Z_{w,p,i,j}\).
  \item \textbf{Canonical output in the decision model:} \\
        \(\forall w,p:\ 
          H_{p,w}=\sum_{i<j}Z_{w,p,i,j}\,i,\quad
          A_{p,w}=\sum_{i<j}Z_{w,p,i,j}\,j\).
  \item \textbf{Output \((H,A)\) from placement/orientation (optimization):}\\
        \(\forall w,p:\ 
          H_{p,w}=\sum_{i<j} Z_{w,p,i,j}\big(i\,B_{i,j}+j(1-B_{i,j})\big),\\
          A_{p,w}=\sum_{i<j} Z_{w,p,i,j}\big(j\,B_{i,j}+i(1-B_{i,j})\big).\)
\end{itemize}

\paragraph{Implied constraints}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \textbf{Exact per–team period profile \(C\):}
        \(\forall t,p:\ C_{t,p}=\sum_{w} U_{w,p,t}\), and
        \(\forall t:\ \sum_{p} C_{t,p}=n-1,\ 
          \sum_{p}[\,C_{t,p}=1\,]=1\).
  \item \textbf{Distinct weeks per team (optimization):}\\
        \(\forall t:\ 
          \texttt{alldifferent}\big([\,W_{\min(t,j),\max(t,j)}\mid j\neq t\,]\big)\)
        (strong global filtering, as recommended by~\cite{HenzMullerThiel2004}).
  \item \textbf{Global home–count (optimization):}
        \(Hc_t=\sum_{j>t}B_{t,j}+\sum_{i<t}(1-B_{i,t})\) and
        \(\sum_{t} Hc_t=\tfrac{n(n-1)}{2}\).
\end{itemize}

\paragraph{Symmetry breaking}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \textbf{Week~1 canonical pairing (both):}
        \(\forall p:\ Z_{1,p,\,2p-1,\,2p}=\texttt{true}\).
  \item \textbf{Week~1 orientation (optimization):}
        \(\forall p:\ B_{2p-1,\,2p}=\texttt{true}\).
  \item \textbf{Single-use period anchor (optimization):}
        \(C_{1,1}=1\).
\end{itemize}

\subsection{Validation}
\label{subsec:cp-validation}

\paragraph{Setup}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \textbf{Instances:} \(n\in\{8,10,12,14,16,18\}\).
  \item \textbf{Solvers:} \texttt{cp-sat}, \texttt{chuffed}, \texttt{gecode}.
  \item \textbf{Time cap:} \(300\) s per run.
  \item \textbf{Approaches:} symmetry breaking (SB) on/off; implied
        constraints (IC) on/off.
  \item \textbf{Search (optimization only):}
        staged search inspired by Trick’s \emph{schedule–then–break}%
        ~\cite{TrickScheduleThenBreak}.
        We separate \emph{structure} from \emph{orientation} and execute
        phases sequentially (no interleaving):\\
        (i) decide the week of each match \(W\) first---this exposes the
        round-robin structure and lets strong globals prune early;\\
        (ii) refine in-week placements \(Z\) once week domains are tight;\\
        (iii) assign home/away \(B\) last, as it chiefly affects fairness.\\
        Heuristics: weighted degree on \(W\) to surface difficult week
        choices; fail-first on the Boolean layers (\(Z,B\)).
  \item \textbf{Result format:}
        decision tables report runtime in seconds (\(>300\) if no answer);
        optimization tables report \emph{time/objective} (objective in
        \textbf{bold} if proven optimal; \(>300/\text{--}\) if no solution).
\end{itemize}

\paragraph{Discussion.}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \textbf{Decision.}
        In the decision version, runs without IC and/or SB often finish sooner
        than the full model with all extras: this task rewards fast steps over
        heavy reasoning. IC adds more checks at each step, and SB can remove
        symmetric solutions that a SAT-based solver might otherwise find quickly.
  \item \textbf{Optimization.}
        With the staged order \(W\rightarrow Z\rightarrow B\), SB and IC help a
        lot: the “weeks all different” and the period-profile constraints narrow
        domains before orientations are chosen, and symmetry breaking removes
        duplicate optimal schedules. Together these effects reduce the work needed
        to prove optimality.
  \item \textbf{Solver contrast.}
        \texttt{cp-sat} is usually the most reliable. Our formulation has large
        Boolean parts (\(Z,U,B\)) with simple linear sums, which fit SAT-based
        solvers that learn from conflicts. This often does better than pure
        finite-domain propagation when the goal is to get a quick solution.
\end{itemize}




\section{SAT Model}

We encode STS in propositional CNF using two families of literals. The solver interface supports Z3 (direct CNF loading) and PySAT backends (DIMACS export with MiniSAT\,22/Glucose\,3). All runs use a $300$s time limit, counted from CNF generation (when applicable), and results are written under \texttt{res/SAT/\#}.json in the format required by the project (\texttt{time}, \texttt{optimal}, \texttt{obj}, \texttt{sol}).

\subsection{Decision variables}

\paragraph{Indexing.}
Let $n$ be even. We use \emph{Teams} $=\{1,\dots,n\}$, \emph{Weeks} $=\{1,\dots,n-1\}$, and \emph{Periods} $=\{1,\dots,n/2\}$. A schedule is an $(n/2)\times(n-1)$ matrix whose entry $(p,w)$ holds exactly one oriented match $(i,j)$ with $i$ at home and $j$ away.

\subsubsection*{Decision variables (literals)}
\begin{itemize}
\item \textbf{Placement with orientation} $X_{i,j,w,p}$ (Boolean): true iff the match between $i$ and $j$ is played in week $w$, period $p$, with $i$ at home (ordered pair, $i\ne j$). This is the authoritative view of the schedule.
\item \textbf{Slot occupancy} $Y_{t,w,p}$ (Boolean): true iff team $t$ occupies slot $(w,p)$. These literals are channeled to $X$ locally (one opponent per team per week).
\end{itemize}


\paragraph{Cardinality encodings.}
We encode $\mathrm{ExactlyOne}$ as $\mathrm{AtLeastOne}+\mathrm{AtMostOne}$, and we choose the CNF for $\mathrm{AtMost}(k)$ adaptively to balance clause/auxiliary-variable overhead and unit-propagation strength. Concretely, for $\mathrm{AtMost}(1)$ we use the pairwise encoding on small scopes ($m\!\le\!8$) because it is aux-free and maximally propagating, whereas we switch to the ladder encoding beyond that to keep the number of clauses linear in $m$. For $k>1$ we adopt sequential counters by default and switch to cardinality networks (\texttt{cardnetwrk}) when the scope becomes large ($m\!\ge\!15$), trading extra auxiliaries for stronger propagation and better scaling.\footnote{These thresholds are the ones used in our implementation.}


\subsection{Objective function}

\subsubsection*{Optimization in SAT (fairness)}
We measure fairness via the maximum home/away deviation per team. Let $T=n-1$ be the number of games per team in a single round robin and $\mathrm{home}_i=\sum_{j\neq i,w,p} X_{i,j,w,p}$. The \emph{home–away balance} definition \cite{knust2006balanced} asks $|\mathrm{home}_i-\mathrm{away}_i|\le1$, which can be simplified to $|\mathrm{home}_i-\tfrac{T}{2}|\le\tfrac12$ for all $i$, i.e., $\mathrm{home}_i\in\{\lfloor T/2\rfloor,\lceil T/2\rceil\}$. We generalize this by minimizing a tolerance $d$:
\[
d^\star \;=\; \min\, d \quad\text{s.t.}\quad
\Big\lceil \tfrac{T-d}{2} \Big\rceil \ \le\ \mathrm{home}_i\ \le\ \Big\lfloor \tfrac{T+d}{2} \Big\rfloor\quad(\forall i).
\]
Parity implies a lower bound $d\ge T\bmod 2$. For fixed $d$, the admissible region is the product of team-wise intervals and thus \emph{monotone} in $d$ (if feasible for $d$, then feasible for any $d'\!>\!d$). Hence $d^\star$ is the smallest $d$ such that the fairness constraints are satisfiable.

\paragraph{SAT-based solution.}
We solve by incremental SAT over $d=1,2,\dots,T$ (binary search is also possible thanks to monotonicity). For each $d$, we create team-wise AtLeast/AtMost cardinalities on \emph{home} literals and test satisfiability. With Z3 we maintain a base context and push/pop only the fairness bounds; with PySAT we export the base CNF once and call a SAT solver with the fairness clauses added on top. This realizes a parametric decision procedure for a minimax objective while reusing learned information across $d$.


\subsection{Constraints}

\subsubsection*{Constraints (decision model)}
All constraints are posted in CNF over two families of literals:
$X_{i,j,w,p}$ and $Y_{t,w,p}$ already explored in section 3.1.

\paragraph{(Global pruning) Week assignment by circle method.}
We pre-assign to every unordered pair $\{u,v\}$ a unique week $w^\star$ by the classical
circle construction. Then we forbid all other weeks with unit negatives and enforce an
\textsf{ExactlyOne} over the literals $\{X_{u,v,w^\star,p},X_{v,u,w^\star,p}\}_{p}$:
\[
\sum_{p}\bigl(X_{u,v,w^\star,p}+X_{v,u,w^\star,p}\bigr)=1,\qquad
X_{u,v,w,p}=X_{v,u,w,p}=0\ \ (\forall w\neq w^\star).
\]
This removes week permutations while preserving at least one feasible schedule.

\paragraph{Channeling $X\leftrightarrow Y$ (one opponent per week).}
Let $o(t,w)$ be the unique opponent of $t$ in week $w$ induced by the circle method. For each $(t,w,p)$:
\[
\bigl(X_{t,o,w,p}\!\!\rightarrow\! Y_{t,w,p}\bigr)\ \wedge\ \bigl(X_{o,t,w,p}\!\!\rightarrow\! Y_{t,w,p}\bigr)\ \wedge\ \bigl(Y_{t,w,p}\!\!\rightarrow\! (X_{t,o,w,p}\vee X_{o,t,w,p})\bigr).
\]
These are 2–3 literal clauses, hence cheap and strong.

\paragraph{Exactly one period per team per week.}
For each $(t,w)$ we post \textsf{ExactlyOne}$\{Y_{t,w,p}\}_p$, which—combined with channeling—
means that each team plays exactly once per week.

\paragraph{Exactly one match per slot.}
For each slot $(w,p)$ we enforce $\sum_t Y_{t,w,p}=2$ via \textsf{AtMost2} and its dual
(\textsf{AtLeast2} encoded as \textsf{AtMost} on negated literals). With channeling, two teams
in a slot imply exactly one oriented $X$ is true there.

\paragraph{Period load cap.}
For each team $t$ and period $p$ we impose $\sum_w Y_{t,w,p}\le 2$ (no team reuses the
same period more than twice over the season).

\subsubsection*{Implied constraints}
(i) Because each pair $\{u,v\}$ is week-fixed, the ``every team plays with every other team only once'' requirement is already
implied by the \textsf{ExactlyOne} inside week $w^\star$ plus the unit negatives for weeks $w\neq w^\star$.
(ii) Variables with $i=j$ are not created, which implies no self-matches.
(iii) ``Exactly two teams per slot'' together with channeling implies ``exactly one match per slot''.

\subsubsection*{Symmetry breaking}
We exploit two levels. (A) \emph{Global}: the circle method fixes the week of each pair, removing
all week permutations. (B) \emph{Local (toggle)}: in week~1 the pair $(i,n\!+\!1\!-\!i)$ is fixed
to period $i$, and match $(1,n)$ is oriented as home for team~1 in $(w,p)=(1,1)$. This removes
residual week/period/orientation symmetries while preserving feasibility.

\subsubsection*{Solvers and interfaces}
\textbf{Z3.} We declare Boolean atoms $v_1,\dots,v_{V}$ and post each CNF clause as a Z3 disjunction; the $300$s limit is passed as a per-call timeout in milliseconds.
\textbf{PySAT backends.} We export CNF to DIMACS (\texttt{pysat.formula.CNF.to\_file}) and run the selected solver with a hard timeout using \texttt{solve\_limited} plus a watchdog \texttt{interrupt}; for optimization we reuse the same DIMACS and add per-$d$ fairness clauses. All solvers are used sequentially, as required.


\paragraph{Why these modelling choices?}
Symmetry breaking and encoding selection strongly affect SAT performance; we deliberately (i) fix a canonical week assignment via the circle method, (ii) keep local $X\!\leftrightarrow\!Y$ channels to obtain short clauses, and (iii) adapt cardinality encodings to scope/threshold. These choices are consistent with the recommendations on the impact of modelling (and symmetry breaking) in SAT problem solving~\cite{martins-impact-sat-modelling}.

\subsection{Validation}

\subsection*{Validation setup}
We evaluate each instance under a fixed wall-clock cap of $300$\,s measured from the start of model/CNF construction to solver termination; the reported time is the total (we also log the build vs.\ solve split for diagnostics). Execution is pinned to a single CPU core.

We use two back-ends: (i) \textbf{Z3}, solving the propositional model in memory; (ii) a \textbf{solver-independent} pipeline that exports DIMACS via PySAT and invokes an external SAT solver selected on the command line (e.g., MiniSAT, Glucose).

For decision runs we report wall-clock runtime. For optimization runs we report the best objective value found within the cap together with an \texttt{optimal} flag.

\paragraph{Observations.}
(i) For $n=16$, the PySAT back-end without extra symmetry breaking (SB) is very fast on the \emph{decision} variant with \textsc{MiniSAT22} (4--6s), whereas \textsc{Glucose3} still times out on the \emph{optimization} variant. A plausible cause is our pipeline: the base CNF is exported once, but optimization tests multiple values of $d$ by spawning a fresh CDCL instance and adding fairness clauses each time (no reuse of learned clauses across $d$), which penalizes solvers more sensitive to restart/learning policies.

(ii) For $n=18$, only \textsc{Z3} solved both decision and optimization within the cap. This matches the fact that Z3 runs in-memory and incrementally: we assert the base CNF once and then push/pop only the fairness bounds (no DIMACS I/O, no solver re-initialization across $d$), which reduces overhead and preserves search state across the $d$-loop.

(iii) Enabling the extra week-1 symmetry sometimes hurts CDCL (e.g., \textsc{MiniSAT22} at $n=16$). This is consistent with known interactions between added SB clauses and branching heuristics: extra structure can constrain the search but also disturb variable activities, worsening the decision order.\cite{martins-impact-sat-modelling}

(iv) Overall, on our model the in-memory incremental path favors \textsc{Z3} as $n$ grows (fewer restarts-from-scratch and cheaper posting of fairness), while the DIMACS+CNF cardinalities in the PySAT path favor \textsc{MiniSAT22} over \textsc{Glucose3} at $n=16$ but degrade under repeated reboots in the optimization loop.


\section{MIP Model}

In this section we present the Mixed-Integer Programming (MIP) formulation adopted for the
Sports Timetabling Scheduling (STS) problem. The model has been implemented in \texttt{Pyomo}
and solved with different solvers (e.g., CBC, HiGHS, and CPLEX). It follows a two-stage
optimization procedure: Stage~A aims at finding a feasible period assignment of all matches,
while Stage~B optimizes the home/away balance for each team given the fixed schedule obtained
from Stage~A. Both stages are linear and solved sequentially within a fixed overall time budget
of 300 seconds.

\subsection{Decision variables}

Let $n$ denote the number of teams (even), $W = n-1$ the number of weeks,
and $P = n/2$ the number of periods per week.  
Let $\mathcal{E}_w = \{(i,j)\ |\ i<j,\ (i,j)\text{ play in week }w\}$ be the set of matches scheduled
in week~$w$ according to a round-robin generation.

\paragraph{Stage A variables.}
\begin{itemize}
    \item $x_{ijwp} \in \{0,1\}$ \quad for all $(i,j,w,p)$ with $(i,j)\in\mathcal{E}_w$ and $p \in \{1,\ldots,P\}$.

    \noindent
    $x_{ijwp} = 1$ if and only if the match $(i,j)$ in week~$w$ is assigned to period~$p$.
\end{itemize}

\paragraph{Stage B variables.}
\begin{itemize}
    \item $s_{ijw} \in \{0,1\}$ \quad for all $(i,j,w)$ with $(i,j)\in\mathcal{E}_w$.

    \noindent
    $s_{ijw} = 1$ if team~$i$ (the lower index) plays \emph{at home} against team~$j$ in week~$w$,
    and $0$ otherwise.
    \item $\mathrm{dev}_i \in \mathbb{R}_{\ge 0}$ \quad for all $i \in \{1,\ldots,n\}$.

    \noindent
    $\mathrm{dev}_i$ measures the deviation of the number of home matches of team~$i$
    from the ideal average $(n-1)/2$.
\end{itemize}

\subsection{Objective function}

\paragraph{Stage A.}  
Stage~A does not include an explicit optimization objective, as it focuses on finding a
feasible assignment of all matches to weeks and periods. The solver is therefore stopped
as soon as a feasible solution is obtained within the allocated time limit.

\paragraph{Stage B.}  
Given a feasible assignment $x^*$ from Stage~A, Stage~B minimizes the total imbalance in
home/away distribution across all teams:
\[
\min \sum_{i=1}^{n} \mathrm{dev}_i
\]
subject to linear inequalities linking $\mathrm{dev}_i$ to the number of home matches:
\[
\begin{aligned}
    \sum_{(i,j,w)\in\mathcal{E}} s_{ijw} +
    \sum_{(j,i,w)\in\mathcal{E}} (1 - s_{jiw}) - \frac{n-1}{2} &\le \mathrm{dev}_i,\\
    \frac{n-1}{2} - \sum_{(i,j,w)\in\mathcal{E}} s_{ijw} -
    \sum_{(j,i,w)\in\mathcal{E}} (1 - s_{jiw}) &\le \mathrm{dev}_i.
\end{aligned}
\]
This ensures that $\mathrm{dev}_i$ captures the absolute deviation of each team’s
home matches from the ideal target.

\subsection{Constraints}

\paragraph{Stage A (period assignment).}
\begin{enumerate}
    \item \textbf{Unique period per match:}
    \[
        \sum_{p=1}^{P} x_{ijwp} = 1
        \quad \forall (i,j,w)\in\mathcal{E}_w
    \]
    Each match in a given week must be scheduled in exactly one period.

    \item \textbf{Unique match per slot:}
    \[
        \sum_{(i,j)\in\mathcal{E}_w} x_{ijwp} = 1
        \quad \forall w \in \{1,\ldots,W\},\ \forall p \in \{1,\ldots,P\}
    \]
    Each slot $(w,p)$ can host at most one match.

    \item \textbf{Period cap constraint:}
    \[
        \sum_{\substack{(a,b,w)\in\mathcal{E}\\ a=i \text{ or } b=i}} x_{abwp} \le 2
        \quad \forall i,\ \forall p
    \]
    Each team can play in the same period at most twice during the whole tournament.

    \item \textbf{Team-week participation:}
    \[
        \sum_{\substack{(a,b,w)\in\mathcal{E}\\ a=i \text{ or } b=i}}
        \sum_{p=1}^{P} x_{abwp} = 1
        \quad \forall i,\ \forall w
    \]
    Each team must play exactly one match per week.

    \item \textbf{Symmetry breaking:}  
    To reduce symmetry among equivalent solutions,
    the first week’s matches are fixed to predefined period indices:
    \[
        x_{ij1p} = 1 \quad \text{for the $p$-th pair in week 1.}
    \]
\end{enumerate}

\paragraph{Stage B (home/away balance).}
Constraints for $\mathrm{dev}_i$ are given in the objective definition above.
An additional symmetry breaking constraint fixes the orientation of the first match:
\[
s_{ij1} = 1 \quad \text{for the first pair $(i,j)$ of week 1.}
\]

\subsection{Validation}

The MIP model was implemented in \texttt{Pyomo} and executed via the
\texttt{SolverFactory} interface.
The default solver used was \texttt{CBC}, with alternative runs using
\texttt{HiGHS} and \texttt{CPLEX} for performance comparison. Note that \texttt{CPLEX} is not bundled in the Docker image due to licensing
restrictions; only \texttt{HiGHS} and \texttt{CBC} are available by default.

A total time budget of 300~seconds was imposed for each instance, divided between
Stage~A and Stage~B according to a user-defined split ratio
($\texttt{splitA}=0.95$ by default).

A greedy warm-start heuristic was employed to initialize feasible values of $x_{ijwp}$
before solving Stage~A. Symmetry-breaking constraints were optionally enabled to
reduce redundant equivalent solutions. Results were exported in JSON format for
subsequent analysis.
